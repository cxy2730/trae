一、Vanguard 反作弊绕过技术
======================================================================

1.1 Vanguard 架构
------------------------------------------------------------------
  Vanguard 是 LoL 自带的内核级反作弊:
  
  进程组成:
  - vgc.exe: Vanguard 服务进程
  - vgk.sys: 内核驱动 (Ring0)
  - League of Legends.exe: 游戏主进程
  
  检测能力:
  1. 内核级内存完整性校验 (PT-D 保护)
  2. 进程黑名单检测 (CreateToolhelp32Snapshot)
  3. 特征码扫描 (扫描已知外挂特征)
  4. 行为模式分析 (API 调用频率/模式)
  5. 网络协议分析 (异常通信检测)
  6. 设备指纹识别 (硬件 ID 绑定)

  2. 在驱动中拦截 Vanguard 检测 API

二、ACE 反作弊绕过技术
======================================================================

2.1 ACE 检测机制
------------------------------------------------------------------
  ACE (AntiCheatExpert) 检测维度:
  
  1. 进程扫描 (CreateToolhelp32Snapshot)
     - 枚举所有运行中进程
     - 比对外挂进程名/窗口标题
     - 检测可疑进程句柄
  
  2. 模块扫描
     - 枚举游戏进程加载的 DLL
     - 比对已知外挂模块签名
     - 检查内存权限异常 (如 EXECUTE_READWRITE)
  
  3. 内存扫描
     - 扫描游戏进程内存空间
     - 特征码 (Signature/AOB) 匹配
     - 代码段完整性校验
  
  4. 行为分析
     - API 调用频率/顺序
     - 线程创建/销毁频率
     - 内存读写模式
  
  5. 网络检测
     - 异常端口通信
     - 可疑协议/加密流量
     - 数据包大小/频率异常

      BOOL result = pOrigFirst(hSnap, pe);
      if (result) {
          // 过滤掉外挂进程
          while (result && IsCheatProcess(pe)) {
              result = pOrigFirst(hSnap, pe);
          }
      }
      return result;
  }

  // 寻找可用的代码洞
  PVOID FindCodeCave(HMODULE hModule, SIZE_T minSize) {
      PIMAGE_DOS_HEADER dos = (PIMAGE_DOS_HEADER)hModule;
      PIMAGE_NT_HEADERS nt = (PIMAGE_NT_HEADERS)((BYTE*)hModule + dos->e_lfanew);
      
      // 遍历节区, 寻找 .text 末尾的对齐空间
      for (WORD i = 0; i < nt->FileHeader.NumberOfSections; i++) {
          PIMAGE_SECTION_HEADER sec = IMAGE_FIRST_SECTION(nt) + i;
          if (sec->Characteristics & IMAGE_SCN_MEM_EXECUTE) {
              DWORD padding = sec->Misc.VirtualSize % 0x1000;
              if (padding > minSize) {
                  return (PVOID)((BYTE*)hModule + sec->VirtualAddress + sec->Misc.VirtualSize);
              }
          }
      }
      return NULL;
  }

  // 变异方法:

  call eax
  
  // 2. 用寄存器中转
  mov ecx, [0x4A258C]
  call ecx
  
  // 3. 用栈中转
  push [0x4A258C]
  call dword ptr [esp]
  add esp, 4
  
  // 4. 运行时生成指令
  BYTE trampoline[] = {
      0xB8, 0x00, 0x00, 0x00, 0x00,  // mov eax, addr
      0xFF, 0xD0                      // call eax
  };
  DWORD targetAddr = 0x4A258C;
  memcpy(&trampoline[1], &targetAddr, 4);
  FlushInstructionCache(GetCurrentProcess(), trampoline, sizeof(trampoline));
  ((VOID(*)())trampoline)();

3.1 API 调用频率控制
------------------------------------------------------------------
  原理: ACE 监控 API 调用频率, 异常高频触发检测
  
  实现: API 调用节流
  
  typedef struct {
      LARGE_INTEGER lastCall;
      DWORD minInterval;  // 最小间隔 (ms)
  } ApiThrottle;
  
  ApiThrottle g_ThrottleTable[] = {
      { {0}, 100 },  // CreateToolhelp32Snapshot: 100ms
      { {0}, 50 },   // OpenProcess: 50ms
      { {0}, 200 },  // WriteProcessMemory: 200ms
      { {0}, 10 },   // GetTickCount: 10ms (高频合法)
  };
  
  BOOL CanCall(LPSTR apiName) {
      LARGE_INTEGER now;
      QueryPerformanceCounter(&now);
      
      for (int i = 0; i < ARRAYSIZE(g_ThrottleTable); i++) {
          if (CompareApiName(apiName, i)) {
              LONGLONG elapsed = (now.QuadPart - 
                  g_ThrottleTable[i].lastCall.QuadPart) / 10000;
              if (elapsed < g_ThrottleTable[i].minInterval) {
                  return FALSE;  // 节流中
              }
              g_ThrottleTable[i].lastCall = now;
              return TRUE;
          }
      }
      return TRUE;
  }

六、进程注入与执行技术
======================================================================

6.1 注入方式选择
------------------------------------------------------------------
  按检测难度排序 (易→难):
  
  1. CreateRemoteThread + LoadLibrary
     - 最常用, 最易检测
     - 特征: 远程线程 + LoadLibraryA
  
  2. WriteProcessMemory + CreateRemoteThread
     - 手动映射 DLL
     - 无 LoadLibrary 调用
  
  3. APC 注入
     - 利用 APC 机制执行
     - 需目标进程处于可警告状态
  
  4. 注册表注入 (AppInit_DLLs)
     - 系统级注入
     - 所有进程都会加载
  
  5. 内核驱动注入
     - Ring0 注入
     - 完全不可检测 (需签名)

6.2 手动映射 DLL (最常用)
------------------------------------------------------------------
  原理: 不使用 LoadLibrary, 手动实现 PE 加载
  
  VOID ManualMap(PIBYTE dllPath) {
      // 1. 读取 DLL 文件
      HANDLE hFile = CreateFileA(dllPath, GENERIC_READ, ...);
      DWORD size = GetFileSize(hFile, NULL);
      BYTE* raw = malloc(size);
      ReadFile(hFile, raw, size, NULL, NULL);
      CloseHandle(hFile);
      
      // 2. 解析 PE
      PIMAGE_DOS_HEADER dos = (PIMAGE_DOS_HEADER)raw;
      PIMAGE_NT_HEADERS nt = (PIMAGE_NT_HEADERS)((BYTE*)raw + dos->e_lfanew);
      
      // 3. 分配目标内存
      BYTE* base = VirtualAlloc(NULL, nt->OptionalHeader.SizeOfImage,
          MEM_COMMIT | MEM_RESERVE, PAGE_EXECUTE_READWRITE);
      
      // 4. 拷贝 PE 头
      memcpy(base, raw, nt->OptionalHeader.SizeOfHeaders);
      
      // 5. 拷贝节区
      for (DWORD i = 0; i < nt->FileHeader.NumberOfSections; i++) {
          PIMAGE_SECTION_HEADER sec = IMAGE_FIRST_SECTION(nt) + i;
          memcpy(base + sec->VirtualAddress,
              raw + sec->PointerToRawData,
              sec->SizeOfRawData);
      }
      
      // 6. 重定位
      DWORD delta = base - nt->OptionalHeader.ImageBase;
      if (delta) ApplyRelocations(base, delta);
      
      // 7. 解析导入
      ResolveImports(base);
      
      // 8. 执行入口点
      typedef BOOL (WINAPI* DllMain_t)(HMODULE, DWORD, LPVOID);
      DllMain_t DllMain = (DllMain_t)(base + nt->OptionalHeader.AddressOfEntryPoint);
      DllMain((HMODULE)base, DLL_PROCESS_ATTACH, NULL);
  }
